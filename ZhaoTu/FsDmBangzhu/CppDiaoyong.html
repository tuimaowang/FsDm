C++ 动态调用 FindPic

业务程序使用与 DLL 配套发布的 FsDmJiekou.h，通过 LoadLibraryW 加载 ZhaoTu.dll，再用 GetProcAddress 获取函数地址。示例中的窗口句柄、查找区域与 DLL 路径必须替换为应用实际值。

#include "FsDmJiekou.h"
#include <windows.h>
#include <cstdint>
#include <cstdio>

using FindPicFn = decltype(&FindPic);
using FreeResultFn = decltype(&FreeResult);
using SetPathFn = decltype(&SetPath);

void SearchPicture(HWND hwnd) {
    HMODULE dll = LoadLibraryW(L"ZhaoTu.dll");
    if (dll == nullptr) {
        std::printf("无法加载 ZhaoTu.dll\n");
        return;
    }

    auto bindWindow = reinterpret_cast<decltype(&BindWindow)>(
        GetProcAddress(dll, "BindWindow"));
    auto unbindWindow = reinterpret_cast<decltype(&UnBindWindow)>(
        GetProcAddress(dll, "UnBindWindow"));
    auto setPath = reinterpret_cast<SetPathFn>(
        GetProcAddress(dll, "SetPath"));
    auto findPic = reinterpret_cast<FindPicFn>(
        GetProcAddress(dll, "FindPic"));
    auto freeResult = reinterpret_cast<FreeResultFn>(
        GetProcAddress(dll, "FreeResult"));

    if (!bindWindow || !unbindWindow || !setPath || !findPic || !freeResult) {
        std::printf("找图接口不完整\n");
        FreeLibrary(dll);
        return;
    }

    // 必须绑定本进程窗口。选择 dx、obs 或 wgc；此处示例使用 wgc。
    if (!bindWindow(hwnd, "wgc", "normal", "normal", 0)) {
        std::printf("窗口绑定失败\n");
        FreeLibrary(dll);
        return;
    }

    setPath(R"(C:\Users\Administrator\Documents\DaMo\FsDm)");
    // 实际程序应先用 GetStatus 确认已获得首帧，再开始找图。
    const FsTongyongJieguo* result =
        findPic(0, 0, 1280, 720, "button.bmp", "000000", 0.9, 0);

    if (result == nullptr) {
        std::printf("结果内存分配失败\n");
    } else {
        if (result->status != FS_CHENGGONG) {
            std::printf("找图失败 status=%d detail=0x%08x\n",
                result->status, static_cast<unsigned int>(result->detail));
        } else if (result->count == 0) {
            std::printf("正常完成，没有找到模板\n");
        } else {
            for (uint32_t i = 0; i < result->count; ++i) {
                const FsJieguoXiang& hit = result->items[i];
                std::printf("id=%d x=%d y=%d file=%s\n",
                    hit.id, hit.x, hit.y, hit.file ? hit.file : "");
            }
        }
        // 结果由 ZhaoTu.dll 分配，必须交还同一个 DLL 释放。
        freeResult(result);
    }

    unbindWindow();
    FreeLibrary(dll);
}

使用时注意

  • HWND 必须是当前进程创建的有效窗口句柄。切换绑定窗口或参数前先解绑。
  • 绑定成功可能仍需等待首帧。实际项目请用 GetStatus 检查 result == FS_CHENGGONG 且 hasFrame != 0 后再查找。
  • DLL 接口采用 C ABI 和 WINAPI 调用约定；函数指针签名必须与发布头文件一致。
  • 读取结果完成后先调用 FreeResult，再卸载 DLL。不要自行 delete、free 结果或内部字段。
  • 示例目录和坐标仅用于演示；路径字符编码为 UTF-8。
若项目启用静态导入而不是运行时动态调用，应按项目构建设置链接对应导入库，并保证运行时能找到 ZhaoTu.dll。不要同时对同一个函数地址使用两种调用方式。
